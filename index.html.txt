<script type="module">
    import { createClient } from 'https://jsdelivr.net'

    // REEMPLAZA ESTOS DOS CAMPOS CON TUS DATOS REALES DE SUPABASE PARA ACTIVAR EL ENVÍO
    const SUPABASE_URL = 'TU_URL_DE_SUPABASE';
    const SUPABASE_ANON_KEY = 'TU_KEY_ANONIMA_DE_SUPABASE';
    const supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

    let tipoActual = 'GENERAL';
    let validadoPorTeclado = false;

    // 1. Geolocalización automática de inicio
    window.addEventListener('DOMContentLoaded', () => {
        if (navigator.geolocation) {
            navigator.geolocation.getCurrentPosition(
                (pos) => {
                    const lat = pos.coords.latitude;
                    const lon = pos.coords.longitude;
                    document.getElementById('inputUbicacion').dataset.lat = lat;
                    document.getElementById('inputUbicacion').dataset.lon = lon;
                    document.getElementById('inputUbicacion').value = `GPS Activo (Lat: ${lat.toFixed(4)}, Lon: ${lon.toFixed(4)})`;
                },
                () => {
                    document.getElementById('inputUbicacion').value = "Chorrillos - Lima";
                }
            );
        }
    });
