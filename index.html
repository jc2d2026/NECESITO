let coordenadasUsuario = null; // BASE DE DATOS LOCALIZADA EN LIMA (Coordenadas aproximadas de Chorrillos y zonas cercanas) const RED_INTERNA_CONFIANZA = [ { nombre: "🔴 Farmacia San Cosme (Sede Chorrillos - Red Interna)", categoria: "farmacia", lat: -12.1855, lng: -77.0215, contacto: "Llamar Central Lima" }, { nombre: "🔴 Técnico Electricista Carlos (Zona Sur - Chorrillos)", categoria: "tecnicos", lat: -12.1910, lng: -77.0180, contacto: "WhatsApp Directo Seguro" }, { nombre: "🔴 Abastos y Víveres Lucía (Mercado Matellini)", categoria: "compras", lat: -12.1820, lng: -77.0150, contacto: "Entrega Prioritaria" } ]; // 1. RECONOCIMIENTO DE VOZ const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; if (SpeechRecognition) { const reconocimiento = new SpeechRecognition(); reconocimiento.lang = 'es-PE'; // Configurado estrictamente para español peruano reconocimiento.continuous = false; document.getElementById('btn-voz-ubicacion').addEventListener('click', () => { reconocimiento.start(); document.getElementById('btn-voz-ubicacion').textContent = "🛑"; reconocimiento.onresult = (event) => { document.getElementById('txt-ubicacion').value = event.results[0][0].transcript; }; reconocimiento.onend = () => { document.getElementById('btn-voz-ubicacion').textContent = "🎙️"; }; }); document.getElementById('btn-voz-pedido').addEventListener('click', () => { reconocimiento.start(); document.getElementById('btn-voz-pedido').textContent = "🛑"; reconocimiento.onresult = (event) => { document.getElementById('txt-pedido').value = event.results[0][0].transcript; }; reconocimiento.onend = () => { document.getElementById('btn-voz-pedido').textContent = "🎙️"; }; }); } // 2. GEOLOCALIZACIÓN INTEGRADA window.addEventListener('load', () => { if (navigator.geolocation) { navigator.geolocation.getCurrentPosition( (position) => { coordenadasUsuario = { lat: position.coords.latitude, lng: position.coords.longitude }; // Muestra al usuario que está validado localmente document.getElementById('txt-ubicacion').value = "GPS: Chorrillos Detectado ✔️"; }, (error) => { document.getElementById('txt-ubicacion').placeholder = "Ingresa tu dirección o usa el micro 🎙️"; }, { enableHighAccuracy: true, timeout: 10000 } ); } }); // FUNCIÓN MATEMÁTICA: Calcular distancia real en kilómetros (Fórmula de Haversine) function calcularDistanciaKm(lat1, lon1, lat2, lon2) { const R = 6371; // Radio de la Tierra en Km const dLat = (lat2 - lat1) * Math.PI / 180; const dLon = (lon2 - lon1) * Math.PI / 180; const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) + Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * Math.sin(dLon / 2) * Math.sin(dLon / 2); const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a)); return R * c; } // 3. PROCESAMIENTO SEGURO CON FILTRADO GEOGRÁFICO document.getElementById('btn-enviar').addEventListener('click', () => { const categoriaSeleccionada = document.getElementById('select-categoria').value; const listaResultados = document.getElementById('lista-resultados'); listaResultados.innerHTML = "Buscando aliados seguros en Chorrillos y alrededores..."; // Filtrar por categoría let resultadosFiltrados = RED_INTERNA_CONFIANZA.filter(item => item.categoria === categoriaSeleccionada); // Si el GPS capturó la ubicación real, medimos la cercanía exacta if (coordenadasUsuario) { resultadosFiltrados = resultadosFiltrados.map(item => { const distancia = calcularDistanciaKm(coordenadasUsuario.lat, coordenadasUsuario.lng, item.lat, item.lng); return { ...item, distancia: distancia }; }); // Ordenar de la más cercana a la más lejana resultadosFiltrados.sort((a, b) => a.distancia - b.distancia); } setTimeout(() => { if (resultadosFiltrados.length > 0) { mostrarEnPantalla("Socios Verificados (Rango Lima Cuadrante Sur)", resultadosFiltrados); } else { // Si está muy lejos o no hay internos, sugiere el mapa de Lima mediante Google Maps corporativo const resultadosGoogleMock = [ { nombre: "🏢 Botica Local Mapeada (Esquina Av. Huaylas / Chorrillos)", contacto: "Ver indicaciones de mapa en Lima" }, { nombre: "🏢 Centro de Salud Materno Infantil de Chorrillos", contacto: "Llamar a Emergencias Locales" } ]; mostrarEnPantalla("Entidades de Respaldo Geográfico (Google Maps Perú)", resultadosGoogleMock); } }, 1000); }); function mostrarEnPantalla(origen, datos) { const listaResultados = document.getElementById('lista-resultados'); listaResultados.innerHTML = `

Origen: ${origen}

`; datos.forEach(item => { const elemento = document.createElement('div'); elemento.className = 'result-item'; // Si calculó distancia, la muestra de forma amigable para el vecino const textoDistancia = item.distancia ? `
📍 A unos ${item.distancia.toFixed(2)} km de tu casa` : ''; elemento.innerHTML = `${item.nombre}${textoDistancia}
Acción: ${item.contacto}`; listaResultados.appendChild(elemento); }); }
